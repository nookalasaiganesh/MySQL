date and time functions
1.CURDATE()-- current date
2.CURTIME()-- current time
3.NOW()-- current date and time
4.DATE-- extracted date part
5.TIME()--extract time part
6.YEAR()--extract year part
7.MONTH()-- extract month
8.DAY()/DAYOFMONTH()-- extract day of the month
9.WEEK()--extract week number
10.DATOFWEEK()--extract day the week
11.DATE_ADD()-- add time to a date
12.DATE_SUB()--subtract time from a date
13.DATEDIFF()-- difference btw two dates
14.TIMESTAMPDIFF()-- difference btw two DATETIME values
15.STR_TO_DATE()-- converts string to date 
16.DATE_FORMAT()
17.LAST_DAY()
18.MAKEDATE()
19.MAKETIME()
20.MONTHNAME()
21.DAYNAME()
22.QUARTER()
23.SEC_TO_TIME()
24.TIME_TO_SEC()
